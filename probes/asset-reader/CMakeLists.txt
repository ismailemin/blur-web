cmake_minimum_required(VERSION 3.25)
project(BlurPakProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_library(pak_reader STATIC src/pak_reader.cpp)
target_include_directories(pak_reader PUBLIC src)
if(MSVC)
  target_compile_options(pak_reader PRIVATE /W4 /permissive-)

  target_compile_options(pak_reader PRIVATE -Wall -Wextra -Wpedantic)
endif()
